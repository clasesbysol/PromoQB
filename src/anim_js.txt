/* ---------- animaciones paso a paso ---------- */
(function(){
  const SP={normal:6500,lento:10500};
  const all=[];
  function setup(a){
    const fr=[...a.querySelectorAll(".an-stage .fr")],caps=[...a.querySelectorAll(".an-cap")],segs=[...a.querySelectorAll(".an-bar button")];
    const ct=a.querySelector(".an-ct"),pb=a.querySelector('[data-a="play"]'),sb=a.querySelector('[data-a="speed"]');
    let i=0,t=null,playing=false,sp="normal";
    function show(k){
      i=Math.max(0,Math.min(fr.length-1,k));
      fr.forEach((f,j)=>{if(j===i){f.classList.remove("on");void f.getBoundingClientRect();f.classList.add("on")}else f.classList.remove("on")});
      caps.forEach((c,j)=>c.classList.toggle("on",j===i));
      segs.forEach((s,j)=>{s.classList.toggle("done",j<i);s.classList.toggle("cur",j===i);s.classList.remove("run")});
      ct.textContent=(i+1)+" / "+fr.length;
      if(playing)tick();
    }
    function tick(){
      clearTimeout(t);a.style.setProperty("--dur",SP[sp]+"ms");
      const s=segs[i];void s.offsetWidth;s.classList.add("run");
      t=setTimeout(()=>{if(i<fr.length-1)show(i+1);else stop()},SP[sp]);
    }
    function play(){all.forEach(o=>o!==api&&o.stop());playing=true;a.classList.add("playing");pb.textContent="❚❚ Pausa";if(i>=fr.length-1)show(0);else show(i)}
    function stop(){playing=false;clearTimeout(t);a.classList.remove("playing");segs.forEach(s=>s.classList.remove("run"));pb.textContent=i>=fr.length-1?"↺ Ver de nuevo":"▶ Reproducir"}
    const api={stop};all.push(api);
    a.addEventListener("click",e=>{
      const b=e.target.closest("button");if(!b||!a.contains(b))return;
      if(b.dataset.i!==undefined){show(+b.dataset.i);if(!playing)stop();return}
      const act=b.dataset.a;
      if(act==="play")playing?stop():play();
      else if(act==="next"){show(i+1);if(!playing)stop()}
      else if(act==="prev"){show(i-1);if(!playing)stop()}
      else if(act==="speed"){sp=sp==="normal"?"lento":"normal";sb.textContent="Velocidad: "+sp;if(playing)tick()}
    });
    const st=a.querySelector(".an-stage");let x0=null;
    st.addEventListener("pointerdown",e=>{x0=e.clientX});
    st.addEventListener("pointerup",e=>{if(x0===null)return;const dx=e.clientX-x0;x0=null;
      if(Math.abs(dx)>45){show(i+(dx<0?1:-1));if(!playing)stop()}});
  }
  document.querySelectorAll(".anim").forEach(setup);
  addEventListener("hashchange",()=>all.forEach(o=>o.stop()));
})();
